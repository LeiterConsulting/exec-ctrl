import sys
import unittest
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[1] / "src"))
from service import coverage, read_invoice


class ServiceTests(unittest.TestCase):
    def test_allowed_invoice(self):
        self.assertEqual(read_invoice("alpha", {"tenant": "alpha", "amount": 42}), 42)

    def test_anonymous_denied(self):
        with self.assertRaisesRegex(PermissionError, "^access denied$"):
            read_invoice(None, {"tenant": "alpha", "amount": 42})

    def test_observed_telemetry(self):
        self.assertEqual(coverage([True, True]), "healthy")
        self.assertEqual(coverage([True, False]), "degraded")


if __name__ == "__main__":
    unittest.main()
